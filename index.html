  <script>

    const yesBtn = document.getElementById("yesBtn");

    const noBtn = document.getElementById("noBtn");

    const success = document.getElementById("success");

    const buttons = document.getElementById("buttons");

    const hint = document.getElementById("hint");

    let noClicks = 0;

    function updateNoButtonText() {

      // Rotate through a few variations

      const texts = [

        "No",

        "Are you sure",

        "Really sure",

        "Last chance",

        "Think again",

        "Do not do this",

        "Ok but why",

        "Still no",

      ];

      noBtn.textContent = texts[Math.min(noClicks, texts.length - 1)];

    }

    function growYesButton() {

      // Make the Yes button larger each time No is clicked

      // Starts at 1.0 and grows up to ~2.6x

      const scale = Math.min(1 + noClicks * 0.2, 2.6);

      yesBtn.style.transform = `scale(${scale})`;

    }

    noBtn.addEventListener("click", () => {

      noClicks++;

      updateNoButtonText();

      growYesButton();

      hint.textContent = "Try again";

    });

    yesBtn.addEventListener("click", () => {

      // Hide buttons, show success section

      buttons.classList.add("hidden");

      hint.classList.add("hidden");

      success.classList.remove("hidden");

    });

  </script>
